# Creating p-values for the Plot

pval_death <- round(cif_fit$Tests["1", "pv"], 3)
pval_transplant <- round(cif_fit$Tests["2", "pv"], 3)
pval_recovery <- round(cif_fit$Tests["3", "pv"], 3)
pval_downgrade <- round(cif_fit$Tests["4","pv"], 3)
pval_upgrade <- round(cif_fit$Tests["5","pv"], 3)
pval_delist <- round(cif_fit$Tests["7","pv"], 3)

# Nicely formats small p-values as e.g. "< 0.001"
format_pval <- function(p) {
  ifelse(p < 0.001, "< 0.001", as.character(round(p, 3)))
}

pval_label <- paste0(
  "Death/Deterioration: p-value = ", format_pval(pval_death), "\n",
  "Transplanted: p-value = ", format_pval(pval_transplant), "\n",
  "Recovery: p-value = ", format_pval(pval_recovery), "\n",
  "Downgrade: p-value=", format_pval(pval_downgrade), "\n",
  "Upgrade: p-value=", format_pval(pval_upgrade),"\n",
  "Delist: p-value=", format_pval(pval_delist)
)
# Plotting the CIF

# Plot the CIF

ggcompetingrisks(
  fit             = cif_fit,
  multiple_panels = FALSE,
  conf.int        = FALSE,
  palette         = c("red", "orange", "yellow", "green","blue","purple","pink"),
  linetype        = c("solid", "dotted"),
  size            = 15
) +
  annotate(
    "text",
    x        = 1,
    y        = 0.95,
    label    = pval_label,
    hjust    = 0,
    vjust    = 1,
    fontface = "bold",
    size     = 3
  ) +
  
  # limits() sets the limits for how many days past being placed in status 2
  scale_x_continuous(breaks = c(0, 10, 20, 30, 40), limits = c(0, 40)) +
  
  scale_y_continuous(breaks = seq(0, 0.9, by = 0.3), limits = c(0, 1)) +
  scale_color_manual(
    values = c("red", "orange","yellow","green","blue","purple","pink"),
    labels = c("Death","Transplanted","Recovery","Downgrade","Upgrade","Censored", "Delist")
  ) +
  scale_linetype_manual(
    values = c("solid", "dotted"),
    labels = c("Stable", "Escalated") #Stable=Solid, Escalated=Dotted
  ) +
  labs(
    x        = "Days since placed in status 2",
    y        = "Cumulative incidence function",
    color    = "Event",
    linetype = "Group"
  ) +
  theme_bw() +
  theme(
    legend.position  = "top",
    legend.text      = element_text(face = "bold", size = 10),
    axis.title       = element_text(face = "bold", size = 10),
    axis.text        = element_text(size = 10),
    panel.grid.minor = element_blank()
  )
